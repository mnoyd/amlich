    ```rust
    use crate::julian::date_to_jd;
    use crate::types::normalize_index;

    /// Calculate signed days from input JD to nearest Solar Term (Tiết Khí)
    ///
    /// Returns:
    /// - 0: Input JD is exactly on a Tiết Khí
    /// - Negative: Nearest Tiết Khí is before input JD
    /// - Positive: Nearest Tiết Khí is after input JD
    ///
    /// # Arguments
    /// * `jd` - Julian Day Number
    ///
    /// # Returns
    /// Signed days difference (negative/zero/positive)
    pub fn get_days_to_nearest_tiet_khi(jd: i32) -> i32 {
        // Get all 24 Tiết Khí JD values from existing constants
        // Reuse existing logic that calculates JD for each term name

        let mut nearest_before: Option<i32> = None;
        let mut nearest_after: Option<i32> = None;

        // Find nearest Tiết Khí before input
        for i in 0..24 {
            let (year, month, day) = JIEQI_DATES[i];
            let tiet_khi_jd = date_to_jd(year, month, day);
            if tiet_khi_jd < jd {
                match nearest_before {
                    None => nearest_before = Some(tiet_khi_jd),
                    Some(ref nearest) if (jd - nearest) < (jd - tiet_khi_jd) => {
                        nearest_before = Some(tiet_khi_jd);
                    }
                    _ => {}
                }
            }
        }

        // Find nearest Tiết Khí after input
        for i in 0..24 {
            let (year, month, day) = JIEQI_DATES[i];
            let tiet_khi_jd = date_to_jd(year, month, day);
            if tiet_khi_jd > jd {
                match nearest_after {
                    None => nearest_after = Some(tiet_khi_jd),
                    Some(ref nearest) if (tiet_khi_jd - jd) < (nearest - jd) => {
                        nearest_after = Some(tiet_khi_jd);
                    }
                    _ => {}
                }
            }
        }

        // Calculate signed difference
        match (nearest_before, nearest_after) {
            (Some(before), None) => before - jd,           // Only before exists
            (None, Some(after)) => after - jd,              // Only after exists
            (Some(before), Some(after)) => {
                // Both exist - pick smaller absolute difference
                let before_diff = (before - jd).abs();
                let after_diff = (after - jd).abs();
                if before_diff <= after_diff {
                    before - jd  // Prefer before when equal
                } else {
                    after - jd
                }
            }
            (None, None) => 0,  // No Tiết Khí found (shouldn't happen)
        }
    }
    ```

    Note: This implementation assumes JIEQI_DATES constant array exists or can be calculated. If not, add helper to calculate JD for each term.
