```rust
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct KnownDivergence {
    pub case: String,            // e.g. "annual 1960"
    pub our_value: u8,
    pub source_values: Vec<SourceValue>,
    pub tiebreaker: String,      // value chosen + which source won (Thẩm Thị Huyền Không Học)
    pub note: String,
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct SourceValue { pub source: String, pub value: u8 }

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PhiTinhGoldenCase {
    pub id: String,
    pub kind: String,            // "annual" | "monthly" | "period"
    pub year: i32,
    #[serde(default)] pub month: Option<u8>,
    #[serde(default)] pub jd: Option<i32>,   // for period boundary cases
    pub van: u8,                 // expected Vận
    pub expected_center: u8,     // expected center star for the layer under test
    pub sources: Vec<SourceValue>,   // >= 2 reference sources
    pub tiebreaker: String,      // Thẩm Thị Huyền Không Học value/citation
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PhiTinhGoldenDataset {
    pub metadata: GoldenMeta,
    pub cases: Vec<PhiTinhGoldenCase>,
    #[serde(default)] pub known_divergences: Vec<KnownDivergence>,
}
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct GoldenMeta { pub description: String, pub case_count: usize }
```
